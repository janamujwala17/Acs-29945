class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int[] result = new int[nums.length];
        for (int i = 0; i <  nums.length; i++){
            for (int j = 0; j < nums.length; j++){
                result[i] += Math.abs(nums[i] - nums[j]);
             }

        }
        return result;
    }
}
