class Solution {
    public int leastInterval(char[] tasks, int n) {
        int[] count = new int[26];
        for (char task : tasks) {
            count[task - 'A']++;
        }
        int max = 0;
        for (int i = 0; i < 26; i++) {
            max = Math.max(max, count[i]);
        }
        int maxCount = 0;
        for (int i = 0; i < 26; i++) {
            if (count[i] == max) {
                maxCount++;
            }
        }
        int result = (max - 1) * (n + 1) + maxCount;
        return Math.max(result, tasks.length);
    }
}
